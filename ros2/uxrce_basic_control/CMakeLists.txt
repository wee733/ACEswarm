cmake_minimum_required(VERSION 3.16)
project(uxrce_basic_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_ros2_cpp REQUIRED)

add_executable(extra_hold src/extra_hold.cpp)
target_compile_features(extra_hold PRIVATE cxx_std_17)
target_link_libraries(extra_hold PRIVATE Eigen3::Eigen px4_ros2_cpp::px4_ros2_cpp rclcpp::rclcpp)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS extra_hold DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/basic_control DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(control_tests test)
endif()
ament_package()
